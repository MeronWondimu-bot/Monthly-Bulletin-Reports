import pandas as pd
import streamlit as st

st.title("ATM Decline Report Automation")

# 1. Upload your multi-sheet Excel file
uploaded_file = st.file_uploader(
    "Upload Daily ATM Decline Report (.xlsx)", type=["xlsx"]
)

if uploaded_file is not None:
  # Read all sheets into a dictionary of DataFrames
  all_sheets = pd.read_excel(uploaded_file, sheet_name=None)

  # Combine all daily sheets into one big DataFrame
  daily_frames = []
  for sheet_name, sheet_df in all_sheets.items():
    # Skip summary sheets if named differently, or process valid daily sheets
    if sheet_name.lower().startswith("sheet"):
      # Clean headers: Assuming row 2 (index 2) contains the headers (RC/BANK NAME, CBE, etc.)
      # We can normalize and stack the data
      sheet_df.columns = sheet_df.iloc[2]
      df_clean = sheet_df.iloc[3:].dropna(
          subset=["RC/BANK NAME"]
      ).copy()  # Filter valid rows
      df_clean["Day_Sheet"] = sheet_name
      daily_frames.append(df_clean)

  if daily_frames:
    combined_df = pd.concat(daily_frames, ignore_index=True)

    st.success(
        f"Successfully combined {len(daily_frames)} daily sheets!"
    )

    # 2. Monthly Summary Option
    st.subheader("Monthly Consolidated Report")
    # Group by Response Code (RC/BANK NAME) and sum numeric bank columns
    numeric_cols = combined_df.select_dtypes(
        include=["number"]
    ).columns.tolist()
    monthly_summary = (
        combined_df.groupby("RC/BANK NAME")[numeric_cols].sum().reset_index()
    )
    st.dataframe(monthly_summary)

    # 3. Bi-Weekly Split Option (e.g., Days 1-15 vs Days 16-31)
    st.subheader("Bi-Weekly Breakdown Option")
    # You can split sheets 1-15 for Bi-Weekly 1 and 16-31 for Bi-Weekly 2
    # Download button for final report
    st.download_button(
        label="Download Monthly Summary as Excel",
        data=monthly_summary.to_csv(index=False),
        file_name="Monthly_ATM_Decline_Summary.csv",
        mime="text/csv",
    )
